Server เริ่มต้น — NewHub() + go hub.Run()

ก่อนที่ client คนแรกจะ connect เข้ามา server สร้าง Hub ขึ้นมา 1 ตัวและเปิด goroutine ให้มัน "เฝ้ารอ" รับ event ตลอดเวลา 

// cmd/server/main.go
hub := internalws.NewHub()   // สร้าง Hub พร้อม channel ทั้งหมด
go hub.Run()              // เปิด goroutine → Hub เริ่ม select loop

// ws_handler.go — Handle()
userID := c.Locals("user_id").(uint)   // จาก JWT middleware
roomID := ParseUint(c.Params("room_id"))

upgrader.Upgrade(c.RequestCtx(), func(conn *websocket.Conn) {
    // ถึงตรงนี้ = upgrade สำเร็จ, conn คือสาย WS ที่เปิดค้างไว้

    client := internalws.NewClient(hub, conn, userID, roomID)
    hub.Register <- client   // ส่งให้ Hub จัดการ

    go client.WritePump()  // goroutine ส่งข้อความออก
    client.ReadPump(...)   // block รับข้อความเข้า
})
NewClient สร้างอะไร?
conn — สายสัญญาณกับ browser  |  send chan — กล่องรับ message จาก Hub  |  UserID, RoomID — identity ของ client คนนี้
2
Hub Register — เพิ่ม Client + แจ้ง Online Status

ทันทีที่ client connect สำเร็จ Hub จะ register และส่ง "ใครออนไลน์อยู่บ้าง" ให้ client ใหม่ทันที พร้อมแจ้งคนในห้องว่ามีคนใหม่เข้ามา 

// ws/hub.go — Run() : case Register
case client := <-h.Register:

    h.rooms[client.RoomID][client] = true        // เพิ่มเข้าห้อง
    h.onlineUsers[client.RoomID][client.UserID]++  // นับ connection

    // 1. ส่ง list คนออนไลน์ให้ client ใหม่คนเดียว
    client.send <- marshalEvent(EventOnlineUsers, ...)

    // 2. broadcast "user_joined" ให้ทุกคนในห้อง
    if h.onlineUsers[roomID][userID] == 1 {   // tab แรก
        h.broadcastToRoom(roomID, marshalEvent(EventUserJoined, ...))
    }

3
ReadPump vs WritePump — 2 goroutine ต่อ client

แต่ละ client มี 2 goroutine ทำงานพร้อมกัน เพราะ "อ่าน" และ "เขียน" ต้องไม่บล็อคกัน

ReadPump
รอรับ message จาก browser ของ client คนนี้

conn.ReadMessage() → block รอ

เมื่อได้รับ:
1. parse JSON
2. save to DB
3. ส่งเข้า hub.broadcast
WritePump
รอรับ message จาก client.send channel

<-c.send → block รอ

เมื่อได้รับ:
1. ส่งออก browser ผ่าน conn.WriteMessage()
2. ping ทุก 54 วิ (keepalive) 
// ws/client.go

func (c *Client) ReadPump(save func(content string) ([]byte, error)) {
    for {
        _, msg, err := c.conn.ReadMessage()  // ← block ตรงนี้
        if err != nil { break }              // browser ปิด → ออก loop

        payload, _ := save(input.Content)
        c.hub.broadcast <- broadcastMsg{roomID: c.RoomID, payload: payload}
    }
    // defer: ส่ง Unregister ให้ Hub
}

func (c *Client) WritePump() {
    for {
        select {
        case msg := <-c.send:                   // ← block ตรงนี้
            c.conn.WriteMessage(TextMessage, msg) // ส่งออก browser
        case <-ticker.C:
            c.conn.WriteMessage(PingMessage, nil)  // keepalive
        }
    }
}

Client Disconnect — Unregister + Cleanup

เมื่อ browser ปิดหรือ network หลุด ReadPump จะออกจาก loop และ cleanup ทุกอย่าง 

// ws/client.go — ReadPump
func (c *Client) ReadPump(...) {
    defer func() {
        c.hub.Unregister <- c  // แจ้ง Hub ว่า client คนนี้หายไปแล้ว
        c.conn.Close()         // ปิด WS connection
    }()

    for {
        _, msg, err := c.conn.ReadMessage()
        if err != nil { break }  // error → browser ปิด → ออก loop → defer รัน
        ...
    }
}
// ws/hub.go — Run() : case Unregister
case client := <-h.Unregister:
    delete(h.rooms[client.RoomID], client)  // เอาออกจากห้อง
    close(client.send)                       // ปิด channel → WritePump จะหยุด

    h.onlineUsers[roomID][userID]--          // ลด connection count
    if count == 0 {                          // ปิดทุก tab แล้ว
        h.broadcastToRoom(roomID, marshalEvent(EventUserLeft, ...))
    }

สรุป — ทุก goroutine ทำงานร่วมกันยังไง
Goroutine Block รอ เมื่อได้รับ
hub.Run() Register / Unregister / broadcast channels อัปเดต rooms map + ส่ง payload ไปที่ client.send
ReadPump (1 ต่อ client) conn.ReadMessage() save DB → ส่งเข้า hub.broadcast
WritePump (1 ต่อ client) <-client.send conn.WriteMessage() ออกไปหา browser
ทำไม Hub ถึง thread-safe โดยไม่ต้องใช้ mutex?
เพราะ rooms และ onlineUsers map ถูกเข้าถึงโดย goroutine เดียวเท่านั้นคือ hub.Run() goroutine อื่นๆ ไม่แตะ map โดยตรง — ส่งทุกอย่างผ่าน channel แทน นี่คือ "share memory by communicating" ซึ่งเป็น Go philosophy
